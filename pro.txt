import "./Profile.css";
import React, { useState } from "react";

function Profile({ person, editPerson, deletePerson }) {

    const [like, setLike] = useState(0);

    const [follow, setFollow] = useState(false);

    function handleClick() {

        setLike(like + 1);

    }

    function handleFollow() {

        setFollow(!follow);

    }

    return (

        <div className="card">

            <h2>{person.name}</h2>

            <h4>{person.role}</h4>

            <p>{person.company}</p>

            <button onClick={handleClick}>

                ❤️ {like}

            </button>

            <button

                onClick={handleFollow}

                className={follow ? "follow-btn followed" : "follow-btn"}

            >

                {follow ? "Following" : "Follow"}

            </button>

            <br /><br />

            <button onClick={() => editPerson(person)}>

                Edit

            </button>

            <button onClick={() => deletePerson(person._id)}>

                Delete

            </button>

        </div>

    );

}

export default Profile;
